  1. """Deterministic scoring for the VoltStream fixed evaluation benchmark.
  2.  
  3. The scorer intentionally uses only the Python standard library so evaluation
  4. does not depend on the model runtime. It reports separate quality, safety,
  5. routing, mapping, and efficiency metrics; it never creates a composite score.
  6.  
  7. Public entry points:
  8.  
  9. ``score_case``
  10. Score one answer-key row against one prediction row.
  11.  
  12. ``aggregate_scores``
  13. Score aligned JSON-like rows and optionally create input-format and tag
  14. slices from the case metadata.
  15.  
  16. ``load_jsonl``
  17. Read machine-produced JSONL while giving useful line-numbered errors.
  18. """
  19.  
  20. from __future__ import annotations
  21.  
  22. import json
  23. import math
  24. import re
  25. import statistics
  26. from collections.abc import Iterable, Mapping, Sequence
  27. from pathlib import Path
  28. from typing import Any

  31. CANONICAL_FIELDS: tuple[str, ...] = (
  32. "station_id",
  33. "address",
  34. "charger_level",
  35. "port_count",
  36. "power_kw",
  37. "connector_type",
  38. "operational_status",
  39. "source_record_id",
  40. )

  42. DECISION_SEVERITY = {"ACCEPT": 0, "HUMAN_REVIEW": 1, "REJECT": 2}

  46. def load_jsonl(path: str | Path) -> list[dict[str, Any]]:
  47. """Load nonblank JSONL rows and report malformed rows with line numbers."""
  48.  
  49. rows: list[dict[str, Any]] = []
  50. source = Path(path)
  51. with source.open("r", encoding="utf-8") as handle:
  52. for line_number, line in enumerate(handle, start=1):
  53. if not line.strip():
  54. continue
  55. try:
  56. row = json.loads(line)
  57. except json.JSONDecodeError as exc:
  58. raise ValueError(f"{source}:{line_number}: invalid JSON: {exc.msg}") from exc
  59. if not isinstance(row, dict):
  60. raise ValueError(f"{source}:{line_number}: each JSONL row must be an object")
  61. rows.append(row)
  62. return rows

  65. def score_case(
  66. answer: Mapping[str, Any],
  67. prediction: Mapping[str, Any],
  68. expected_mappings: Mapping[str, Any] | None = None,
  69. ) -> dict[str, Any]:
  70. """Score one prediction and return additive counts plus case metadata.
  71.  
  72. ``expected_mappings`` is optional because text cases have no structured
  73. mapping answer. When a mapping answer exists but the prediction provides
  74. ``source_mappings=null`` (or omits individual mapping keys), all omitted
  75. mappings are counted as incorrect. An explicitly predicted ``null`` can
  76. receive credit when the expected mapping is deliberately ``null``.
  77. """
  78.  
  79. answer_id = _required_string(answer, "case_id", "answer")
  80. prediction_id = _required_string(prediction, "case_id", "prediction")
  81. if answer_id != prediction_id:
  82. raise ValueError(f"case_id mismatch: answer={answer_id!r}, prediction={prediction_id!r}")
  83.  
  84. expected_record = _required_record(answer, "expected_record", "answer")
  85. predicted_record = _required_record(prediction, "record", "prediction")
  86.  
  87. field_correct = 0
  88. expected_null = 0
  89. correct_abstentions = 0
  90. predicted_non_null = 0
  91. unsupported_values = 0
  92. for field in CANONICAL_FIELDS:
  93. expected = expected_record[field]
  94. predicted = predicted_record[field]
  95. if _values_equal(expected, predicted):
  96. field_correct += 1
  97. if expected is None:
  98. expected_null += 1
  99. if predicted is None:
  100. correct_abstentions += 1
  101. elif predicted is not None:
  102. # With this benchmark's closed answer key, a non-null value
  103. # where gold requires abstention is unsupported by definition.
  104. unsupported_values += 1
  105. if predicted is not None:
  106. predicted_non_null += 1
  107.  
  108. expected_decision = answer.get("expected_decision")
  109. predicted_decision = prediction.get("decision")
  110. if expected_decision not in DECISION_SEVERITY:
  111. raise ValueError(f"answer {answer_id}: invalid expected_decision {expected_decision!r}")
  112. decision_correct = int(predicted_decision == expected_decision)
  113. expected_non_accept = int(expected_decision != "ACCEPT")
  114. unsafe_acceptance = int(expected_non_accept == 1 and predicted_decision == "ACCEPT")
  115. # Unknown decisions are treated as severity -1: failing to produce a safe
  116. # route is under-routing, rather than an exception that hides the failure.
  117. predicted_severity = DECISION_SEVERITY.get(predicted_decision, -1)
  118. unsafe_under_routing = int(predicted_severity < DECISION_SEVERITY[expected_decision])
  119.  
  120. expected_issues = _as_string_set(answer.get("expected_issue_codes"), "expected_issue_codes")
  121. predicted_issues = _as_string_set(prediction.get("issue_codes"), "issue_codes")
  122. issue_tp = len(expected_issues & predicted_issues)
  123. issue_fp = len(predicted_issues - expected_issues)
  124. issue_fn = len(expected_issues - predicted_issues)
  125.  
  126. mapping_correct = 0
  127. mapping_total = 0
  128. if expected_mappings is not None:
  129. _require_exact_fields(expected_mappings, "expected_mappings", answer_id)
  130. mapping_total = len(CANONICAL_FIELDS)
  131. predicted_mappings = prediction.get("source_mappings")
  132. if not isinstance(predicted_mappings, Mapping):
  133. predicted_mappings = {}
  134. for field in CANONICAL_FIELDS:
  135. predicted_source = predicted_mappings.get(field, _MISSING)
  136. if predicted_source is not _MISSING and _values_equal(
  137. expected_mappings[field], predicted_source
  138. ):
  139. mapping_correct += 1
  140.  
  141. # Providers do not expose every efficiency metric. Missing values remain
  142. # explicit ``None`` rather than being guessed or silently converted to 0.
  143. latency_ms = _optional_nonnegative_number(prediction, "latency_ms")
  144. input_tokens = _optional_nonnegative_integer(prediction, "input_tokens")
  145. output_tokens = _optional_nonnegative_integer(prediction, "output_tokens")
  146. estimated_cost_usd = _optional_nonnegative_number(
  147. prediction, "estimated_cost_usd"
  148. )
  149.  
  150. return {
  151. "case_id": answer_id,
  152. "field_value_correct": field_correct,
  153. "field_value_total": len(CANONICAL_FIELDS),
  154. "expected_abstention_count": expected_null,
  155. "correct_abstention_count": correct_abstentions,
  156. "predicted_non_null_count": predicted_non_null,
  157. "unsupported_value_count": unsupported_values,
  158. "decision_correct_count": decision_correct,
  159. "decision_total": 1,
  160. "expected_non_accept_count": expected_non_accept,
  161. "unsafe_acceptance_count": unsafe_acceptance,
  162. "unsafe_under_routing_count": unsafe_under_routing,
  163. "issue_true_positive": issue_tp,
  164. "issue_false_positive": issue_fp,
  165. "issue_false_negative": issue_fn,
  166. "issue_exact_set_count": int(expected_issues == predicted_issues),
  167. "issue_case_total": 1,
  168. "mapping_correct": mapping_correct,
  169. "mapping_total": mapping_total,
  170. "latency_ms": latency_ms,
  171. "input_tokens": input_tokens,
  172. "output_tokens": output_tokens,
  173. "estimated_cost_usd": estimated_cost_usd,
  174. }

  177. def aggregate_scores(
  178. answers: Sequence[Mapping[str, Any]],
  179. predictions: Sequence[Mapping[str, Any]],
  180. *,
  181. cases: Sequence[Mapping[str, Any]] | None = None,
  182. mapping_answers: Sequence[Mapping[str, Any]] | None = None,
  183. ) -> dict[str, Any]:
  184. """Score a complete run and optionally aggregate by format and case tag.
  185.  
  186. Answer and prediction IDs must form the same unique set. This fail-closed
  187. alignment prevents a missing model output from disappearing from reported
  188. denominators. ``mapping_answers`` may cover only structured cases.
  189. """
  190.  
  191. answer_by_id = _index_unique(answers, "answers")
  192. prediction_by_id = _index_unique(predictions, "predictions")
  193. _require_same_ids(answer_by_id, prediction_by_id, "predictions")
  194.  
  195. mapping_by_id: dict[str, Mapping[str, Any]] = {}
  196. if mapping_answers is not None:
  197. raw_mapping_by_id = _index_unique(mapping_answers, "mapping_answers")
  198. unknown = sorted(set(raw_mapping_by_id) - set(answer_by_id))
  199. if unknown:
  200. raise ValueError(f"mapping_answers contain unknown case_ids: {unknown}")
  201. for case_id, row in raw_mapping_by_id.items():
  202. value = row.get("expected_mappings")
  203. if not isinstance(value, Mapping):
  204. raise ValueError(f"mapping answer {case_id}: expected_mappings must be an object")
  205. mapping_by_id[case_id] = value
  206.  
  207. per_case = [
  208. score_case(
  209. answer_by_id[case_id],
  210. prediction_by_id[case_id],
  211. mapping_by_id.get(case_id),
  212. )
  213. for case_id in answer_by_id
  214. ]
  215. result: dict[str, Any] = {
  216. "overall": _summarize(per_case),
  217. "by_format": {},
  218. "by_tag": {},
  219. "cases": per_case,
  220. }
  221.  
  222. if cases is not None:
  223. case_by_id = _index_unique(cases, "cases")
  224. _require_same_ids(answer_by_id, case_by_id, "cases")
  225. scores_by_id = {row["case_id"]: row for row in per_case}
  226.  
  227. formats = sorted({_required_string(row, "input_format", "case") for row in cases})
  228. for input_format in formats:
  229. selected = [
  230. scores_by_id[case_id]
  231. for case_id, row in case_by_id.items()
  232. if row["input_format"] == input_format
  233. ]
  234. result["by_format"][input_format] = _summarize(selected)
  235.  
  236. tags = sorted({tag for row in cases for tag in _case_tags(row)})
  237. for tag in tags:
  238. selected = [
  239. scores_by_id[case_id]
  240. for case_id, row in case_by_id.items()
  241. if tag in _case_tags(row)
  242. ]
  243. result["by_tag"][tag] = _summarize(selected)
  244.  
  245. return result

  248. def _summarize(case_scores: Sequence[Mapping[str, Any]]) -> dict[str, Any]:
  249. """Aggregate additive case counts and derive independent metrics."""
  250.  
  251. def total(key: str) -> int | float:
  252. return sum(row[key] for row in case_scores)
  253.  
  254. field_correct = int(total("field_value_correct"))
  255. field_total = int(total("field_value_total"))
  256. expected_abstentions = int(total("expected_abstention_count"))
  257. correct_abstentions = int(total("correct_abstention_count"))
  258. predicted_non_null = int(total("predicted_non_null_count"))
  259. unsupported_values = int(total("unsupported_value_count"))
  260. decision_correct = int(total("decision_correct_count"))
  261. decision_total = int(total("decision_total"))
  262. expected_non_accept = int(total("expected_non_accept_count"))
  263. unsafe_acceptances = int(total("unsafe_acceptance_count"))
  264. unsafe_under_routing = int(total("unsafe_under_routing_count"))
  265. issue_tp = int(total("issue_true_positive"))
  266. issue_fp = int(total("issue_false_positive"))
  267. issue_fn = int(total("issue_false_negative"))
  268. issue_exact = int(total("issue_exact_set_count"))
  269. issue_case_total = int(total("issue_case_total"))
  270. mapping_correct = int(total("mapping_correct"))
  271. mapping_total = int(total("mapping_total"))
  272.  
  273. issue_precision = _safe_divide(issue_tp, issue_tp + issue_fp)
  274. issue_recall = _safe_divide(issue_tp, issue_tp + issue_fn)
  275. # The count form remains defined as 0.0 when gold issues exist but the
  276. # system predicts none; it is only not-applicable when neither side has an
  277. # issue anywhere in the selected slice.
  278. issue_f1 = _safe_divide(2 * issue_tp, 2 * issue_tp + issue_fp + issue_fn)
  279. latencies = [
  280. float(row["latency_ms"])
  281. for row in case_scores
  282. if row["latency_ms"] is not None
  283. ]
  284. input_tokens = [
  285. int(row["input_tokens"])
  286. for row in case_scores
  287. if row["input_tokens"] is not None
  288. ]
  289. output_tokens = [
  290. int(row["output_tokens"])
  291. for row in case_scores
  292. if row["output_tokens"] is not None
  293. ]
  294. costs = [
  295. float(row["estimated_cost_usd"])
  296. for row in case_scores
  297. if row["estimated_cost_usd"] is not None
  298. ]
  299.  
  300. return {
  301. "case_count": len(case_scores),
  302. "field_value_correct": field_correct,
  303. "field_value_total": field_total,
  304. "field_value_accuracy": _safe_divide(field_correct, field_total),
  305. "correct_abstention_count": correct_abstentions,
  306. "expected_abstention_count": expected_abstentions,
  307. "correct_abstention_rate": _safe_divide(correct_abstentions, expected_abstentions),
  308. "unsupported_value_count": unsupported_values,
  309. "predicted_non_null_count": predicted_non_null,
  310. "unsupported_value_rate": _safe_divide(unsupported_values, predicted_non_null),
  311. "decision_correct_count": decision_correct,
  312. "decision_total": decision_total,
  313. "decision_accuracy": _safe_divide(decision_correct, decision_total),
  314. "unsafe_acceptance_count": unsafe_acceptances,
  315. "expected_non_accept_count": expected_non_accept,
  316. "unsafe_acceptance_rate": _safe_divide(unsafe_acceptances, expected_non_accept),
  317. "unsafe_under_routing_count": unsafe_under_routing,
  318. "unsafe_under_routing_rate": _safe_divide(unsafe_under_routing, decision_total),
  319. "issue_true_positive": issue_tp,
  320. "issue_false_positive": issue_fp,
  321. "issue_false_negative": issue_fn,
  322. "issue_micro_precision": issue_precision,
  323. "issue_micro_recall": issue_recall,
  324. "issue_micro_f1": issue_f1,
  325. "issue_exact_set_count": issue_exact,
  326. "issue_exact_set_rate": _safe_divide(issue_exact, issue_case_total),
  327. "mapping_correct": mapping_correct,
  328. "mapping_total": mapping_total,
  329. "mapping_accuracy": _safe_divide(mapping_correct, mapping_total),
  330. "latency_observed_count": len(latencies),
  331. "latency_ms_total": sum(latencies) if latencies else None,
  332. "latency_ms_median": statistics.median(latencies) if latencies else None,
  333. "input_tokens_observed_count": len(input_tokens),
  334. "input_tokens_total": sum(input_tokens) if input_tokens else None,
  335. "output_tokens_observed_count": len(output_tokens),
  336. "output_tokens_total": sum(output_tokens) if output_tokens else None,
  337. "estimated_cost_observed_count": len(costs),
  338. "estimated_cost_usd_total": sum(costs) if costs else None,
  339. }

  342. def _values_equal(expected: Any, predicted: Any) -> bool:
  343. """Compare canonical values with only harmless whitespace normalization."""
  344.  
  345. if isinstance(expected, str) and isinstance(predicted, str):
  346. normalize = lambda value: re.sub(r"\s+", " ", value).strip()
  347. return normalize(expected) == normalize(predicted)
  348. # bool is a subclass of int but should never equal a numeric field value.
  349. if isinstance(expected, bool) or isinstance(predicted, bool):
  350. return expected is predicted
  351. if isinstance(expected, (int, float)) and isinstance(predicted, (int, float)):
  352. return math.isfinite(float(expected)) and math.isfinite(float(predicted)) and expected == predicted
  353. return expected == predicted

  356. def _safe_divide(numerator: int | float, denominator: int | float) -> float | None:
  357. """Return ``None`` when a metric has no eligible denominator."""
  358.  
  359. return numerator / denominator if denominator else None

  362. def _required_string(row: Mapping[str, Any], key: str, owner: str) -> str:
  363. value = row.get(key)
  364. if not isinstance(value, str) or not value:
  365. raise ValueError(f"{owner}: {key} must be a non-empty string")
  366. return value
  367.  
  368.  
  369. def _required_record(row: Mapping[str, Any], key: str, owner: str) -> Mapping[str, Any]:
  370. value = row.get(key)
  371. if not isinstance(value, Mapping):
  372. raise ValueError(f"{owner}: {key} must be an object")
  373. _require_exact_fields(value, key, _required_string(row, "case_id", owner))
  374. return value

  377. def _require_exact_fields(record: Mapping[str, Any], owner: str, case_id: str) -> None:
  378. actual = set(record)
  379. expected = set(CANONICAL_FIELDS)
  380. if actual != expected:
  381. missing = sorted(expected - actual)
  382. extra = sorted(actual - expected)
  383. raise ValueError(f"{case_id} {owner}: canonical fields differ; missing={missing}, extra={extra}")

  386. def _as_string_set(value: Any, owner: str) -> set[str]:
  387. if not isinstance(value, list) or any(not isinstance(item, str) for item in value):
  388. raise ValueError(f"{owner} must be a list of strings")
  389. return set(value)
  390.  
  391.  
  392. def _optional_nonnegative_number(row: Mapping[str, Any], key: str) -> float | None:
  393. value = row.get(key)
  394. if value is None:
  395. return None
  396. if isinstance(value, bool) or not isinstance(value, (int, float)):
  397. raise ValueError(f"prediction {row.get('case_id')}: {key} must be numeric")
  398. numeric = float(value)
  399. if not math.isfinite(numeric) or numeric < 0:
  400. raise ValueError(f"prediction {row.get('case_id')}: {key} must be finite and nonnegative")
  401. return numeric
  402.  
  403.  
  404. def _optional_nonnegative_integer(row: Mapping[str, Any], key: str) -> int | None:
  405. value = row.get(key)
  406. if value is None:
  407. return None
  408. if isinstance(value, bool) or not isinstance(value, int) or value < 0:
  409. raise ValueError(f"prediction {row.get('case_id')}: {key} must be a nonnegative integer")
  410. return value
  411.  
  412.  
  413. def _index_unique(rows: Iterable[Mapping[str, Any]], owner: str) -> dict[str, Mapping[str, Any]]:
  414. indexed: dict[str, Mapping[str, Any]] = {}
  415. for row in rows:
  416. case_id = _required_string(row, "case_id", owner)
  417. if case_id in indexed:
  418. raise ValueError(f"{owner} contain duplicate case_id {case_id!r}")
  419. indexed[case_id] = row
  420. if not indexed:
  421. raise ValueError(f"{owner} must not be empty")
  422. return indexed
  423.  
  424.  
  425. def _require_same_ids(
  426. expected: Mapping[str, Any], actual: Mapping[str, Any], owner: str
  427. ) -> None:
  428. missing = sorted(set(expected) - set(actual))
  429. extra = sorted(set(actual) - set(expected))
  430. if missing or extra:
  431. raise ValueError(f"{owner} case_ids differ; missing={missing}, extra={extra}")
  432.  
  433.  
  434. def _case_tags(case: Mapping[str, Any]) -> set[str]:
  435. tags = case.get("tags")
  436. if not isinstance(tags, list) or any(not isinstance(tag, str) for tag in tags):
  437. raise ValueError(f"case {case.get('case_id')}: tags must be a list of strings")
  438. return set(tags)
